              Quickstart
              
from openai import OpenAI

client = OpenAI(
    api_key="YOUR_KEY",
    base_url="https://llm.maruthlabs.com/v1"
)
r = client.chat.completions.create(
    model="madhuram",
    messages=[{"role": "user", "content": "Hello"}]
)
print(r.choices[0].message.content)
